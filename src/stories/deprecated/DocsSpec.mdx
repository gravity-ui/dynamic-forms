import {Meta, Markdown} from '@storybook/addon-docs';

import Spec from '../../../docs/deprecated/spec.md?raw';

<Meta title="Deprecated/Docs/Spec" />

export const replacements = [
    ['./lib.md', '?path=/docs/deprecated-docs-lib--docs'],
    ['./config.md', '?path=/docs/deprecated-docs-config--docs'],
    ['./input-props-map.md', '?path=/docs/deprecated-docs-input-props-map--docs'],
];

export const applyReplacements = (text, replacements) => {
    return replacements.reduce((result, [searchValue, replaceValue]) => {
        return result.split(searchValue).join(replaceValue);
    }, text);
};

export const content = applyReplacements(Spec, replacements);

<Markdown>{content}</Markdown>
